# Project-level configuration.
cmake_minimum_required(VERSION 3.13)
project(runner LANGUAGES CXX)
set(BINARY_NAME "boomprint")
set(APPLICATION_ID "com.rnd.boomprint")

set(_LINUX_GTK_VARIANT "$ENV{FLUTTER_LINUX_GTK}")
if("${_LINUX_GTK_VARIANT}" STREQUAL "")
  set(_LINUX_GTK_VARIANT_FILE "${CMAKE_CURRENT_SOURCE_DIR}/.gtk_variant.cmake")
  if(EXISTS "${_LINUX_GTK_VARIANT_FILE}")
    include("${_LINUX_GTK_VARIANT_FILE}")
    set(_LINUX_GTK_VARIANT "${LINUX_GTK_VARIANT}")
  elseif(DEFINED LINUX_GTK_VARIANT)
    set(_LINUX_GTK_VARIANT "${LINUX_GTK_VARIANT}")
  else()
    set(_LINUX_GTK_VARIANT "gtk3")
  endif()
endif()
set(LINUX_GTK_VARIANT "${_LINUX_GTK_VARIANT}" CACHE STRING "Linux GTK runner variant" FORCE)
set_property(CACHE LINUX_GTK_VARIANT PROPERTY STRINGS "gtk3" "gtk4")

if("${LINUX_GTK_VARIANT}" STREQUAL "gtk4")
  set(LINUX_GTK_PKG gtk4)
  set(LINUX_GTK_DEFINE FLUTTER_LINUX_GTK4)
elseif("${LINUX_GTK_VARIANT}" STREQUAL "gtk3")
  set(LINUX_GTK_PKG gtk+-3.0)
  set(LINUX_GTK_DEFINE FLUTTER_LINUX_GTK3)
else()
  message(FATAL_ERROR "Unsupported LINUX_GTK_VARIANT: ${LINUX_GTK_VARIANT}. Expected gtk3 or gtk4.")
endif()

message(STATUS "Configuring Linux runner for ${LINUX_GTK_VARIANT} (${LINUX_GTK_PKG})")

cmake_policy(SET CMP0063 NEW)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/linux_project_common.cmake")
